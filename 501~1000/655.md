# 655. Print Binary Tree
## Approach
1. Calculate the height
2. DFS
3. update `res` based on the problem description.
## Code
```cpp
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
    public: int h,
    m,
    n;
    int height(TreeNode * root) {
        if (root == nullptr) return -1;
        int lheight = height(root -> left);
        int rheight = height(root -> right);
        return max(lheight, rheight) + 1;
    }
    void dfs(TreeNode * root, int r, int c, vector < vector < string >> & res) {
        res[r][c] = to_string(root -> val);
        if (root -> left) dfs(root -> left, r + 1, c - (1 << (h - r - 1)), res);
        if (root -> right) dfs(root -> right, r + 1, c + (1 << (h - r - 1)), res);
    }
    vector < vector < string >> printTree(TreeNode * root) {
        h = height(root);
        m = h + 1;
        n = (1 << m) - 1;
        vector < vector < string >> res(m, vector < string > (n, ""));
        dfs(root, 0, (n - 1) / 2, res);
        return res;
    }
};
```
